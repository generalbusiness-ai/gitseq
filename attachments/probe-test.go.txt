package app

import (
 "context"
 "strings"
 "testing"
 "github.com/generalbusiness-ai/gitseq/internal/workroom"
 "github.com/generalbusiness-ai/gitseq/internal/apphost"
)

func TestPlannerNoLocalCeilingSignsOversizedBeforeKernelRefusal(t *testing.T) {
 ctx := context.Background()
 repo := testRepo(t)
 w,seed,err := Init(ctx,repo,"human",2<<10)
 if err != nil {t.Fatal(err)}
 // Simulate a legacy fixture config with no recorded ceiling, then reopen it.
 // Preserve every custody and verified-frontier field; genesis remains 2048.
 cfg := w.config.Clone()
 cfg.PayloadCeiling=0
 if err:=apphost.SaveConfig(w.MetaDir,cfg);err!=nil {t.Fatal(err)}
 w,err=Open(ctx,repo);if err!=nil {t.Fatal(err)}
 counter := &signatureCounter{}
 w.SetObserver(counter)
 before,err := w.Snapshot(ctx);if err != nil {t.Fatal(err)}
 _,err=w.Act(ctx,"human",Act{Verb:VerbState,Kind:workroom.KindAssert,Text:"oversized with no local ceiling",RestsOn:[]string{seed.ID},Attachments:map[string][]byte{"evidence.txt":make([]byte,4<<10)},IdempotencyKey:"planner-zero-ceiling-probe"})
 if err==nil || !strings.Contains(err.Error(),"event exceeds genesis ceiling") {t.Fatalf("unexpected kernel result: %v",err)}
 if counter.count.Load()!=1 {t.Fatalf("observed signatures=%d, expected current bypass to sign once",counter.count.Load())}
 actErr := err
 after,err:=w.Snapshot(ctx);if err!=nil {t.Fatal(err)}
 if after.Head!=before.Head || after.Depth!=before.Depth {t.Fatal("kernel refusal changed the frontier")}
 t.Logf("reproduced missing-local-ceiling gap: signatures=%d; frontier unchanged=%v; error=%v",counter.count.Load(),after.Head==before.Head,actErr)
}
