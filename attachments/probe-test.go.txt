// Append this function to cmd/gitseq-perf/main_test.go through a temporary go test -overlay. Build ./cmd/gitseq-perf under Go1.26.7 and pass its absolute path in PLANNER_PERF_PROBE_BINARY. Run only this test with -count=1 -v. No production source edits.
func TestPlannerWorkerDiagnosticStderrViability(t *testing.T) {
 ctx := context.Background()
 fixture := filepath.Join(t.TempDir(), "fixture")
 _, err := perfscenario.Prepare(ctx, fixture, perfscenario.FixturePlan{GeneratorVersion:"planner-stderr-proof.v1",Seed:632,Depth:10,Shape:"linear",PayloadBuckets:[]int{8},ActorCount:1})
 if err != nil { t.Fatal(err) }
 selected := runCase{Scenario:"cold_status",Shape:"linear",Depth:10,Tail:-1,Concurrency:1,ActorCount:1,Fanout:1}
 binary := os.Getenv("PLANNER_PERF_PROBE_BINARY")
 if binary == "" { t.Fatal("missing probe binary") }
 t.Setenv("GODEBUG", "")
 result, err := runWorker(ctx,binary,fixture,selected,testContract(t),"",false)
 if err != nil { t.Fatalf("ordinary worker: %v",err) }
 t.Logf("ordinary worker succeeded: schema=%s depth=%d",result.Schema,result.Depth)
 t.Setenv("GODEBUG", "gctrace=1")
 _, err = runWorker(ctx,binary,fixture,selected,testContract(t),"",false)
 if err == nil { t.Fatal("gctrace worker unexpectedly decoded despite current CombinedOutput") }
 if !strings.Contains(err.Error(),"decode worker result") || !strings.Contains(err.Error(),"gc ") { t.Fatalf("unexpected failure: %v",err) }
 t.Logf("reproduced successful-process stderr contamination: %v",err)
}
