import { readFileSync } from "node:fs";
import React from "react";
import { renderToStaticMarkup } from "react-dom/server";
import { createServer } from "vite";
const [uiRoot, docPath, head, event, expectUrl] = process.argv.slice(2);
const text = readFileSync(docPath, "utf8");
const vite = await createServer({ root: uiRoot, appType: "custom", logLevel: "silent", server: { middlewareMode: true } });
try {
  const { MarkdownPreview, PreviewContext } = await vite.ssrLoadModule("/src/components/Preview.tsx");
  const html = renderToStaticMarkup(React.createElement(PreviewContext.Provider, { value: { address: { kind: "notes", event }, open() {} } },
    React.createElement(MarkdownPreview, { event, commit: head, path: "docs/reference/reading-view.md", text })));
  const anchors = [...html.matchAll(/<a href="([^"]+)"[^>]*>(.*?)<\/a>/g)].map(m => ({ href: m[1].replace(/&amp;/g, "&"), label: m[2] }));
  const note = "notes/2026-09-07-previous-frontier-readers.md";
  const native = anchors.filter(a => a.href.startsWith("#/")).map(a => { const q = new URLSearchParams(a.href.slice(a.href.indexOf("?") + 1)); return { label: a.label, file: q.get("file"), at: q.get("at"), line: q.get("line"), event: q.get("preview_event") }; });
  const noteNative = native.filter(p => p.file === note);
  const ext = anchors.filter(a => a.href === expectUrl && a.label === "Previous-frontier readers during a rebuild");
  const unavailableNear = (html.match(/Previous-frontier readers during a rebuild[^<]*<span[^>]*>\(link unavailable\)/g) || []).length;
  const unavailable = (html.match(/link unavailable/g) || []).length;
  const ok = noteNative.length === 1 && noteNative[0].at === head && noteNative[0].event === event && !noteNative[0].line && ext.length === 1 && unavailableNear === 0;
  console.log(JSON.stringify({ head, noteNativeAnchors: noteNative.length, noteNative: noteNative[0] || null, externalNoteAnchor: ext.length, unavailableAtThisLink: unavailableNear, unavailableTotal: unavailable, anchorsTotal: anchors.length, ok }, null, 1));
  process.exitCode = ok ? 0 : 1;
} finally { await vite.close(); }
