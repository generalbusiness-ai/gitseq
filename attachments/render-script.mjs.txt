import { readFileSync } from "node:fs";
import React from "react";
import { renderToStaticMarkup } from "react-dom/server";
import { createServer } from "vite";
const [uiRoot, notePath, head, event] = process.argv.slice(2);
const text = readFileSync(notePath, "utf8");
const vite = await createServer({ root: uiRoot, appType: "custom", logLevel: "silent", server: { middlewareMode: true } });
try {
  const { MarkdownPreview, PreviewContext } = await vite.ssrLoadModule("/src/components/Preview.tsx");
  const html = renderToStaticMarkup(React.createElement(PreviewContext.Provider, { value: { address: { kind: "notes", event }, open() {} } },
    React.createElement(MarkdownPreview, { event, commit: head, path: "notes/2026-09-07-previous-frontier-readers.md", text })));
  const anchors = [...html.matchAll(/<a href="([^"]+)"[^>]*>(.*?)<\/a>/g)].map(m => ({ href: m[1].replace(/&amp;/g, "&"), label: m[2] }));
  const previews = anchors.filter(a => a.href.startsWith("#/")).map(a => { const q = new URLSearchParams(a.href.slice(a.href.indexOf("?") + 1)); return { label: a.label, file: q.get("file"), at: q.get("at"), line: q.get("line"), event: q.get("preview_event") }; });
  const external = anchors.filter(a => !a.href.startsWith("#/"));
  const expected = [...text.matchAll(/\[([^\]]+)\]\((https:\/\/github\.com\/generalbusiness-ai\/gitseq\/blob\/([0-9a-f]{40})\/([^)#]+)#L(\d+))\), `\4:\5`/g)].map(m => ({ label: m[1], url: m[2], pin: m[3], file: m[4], line: m[5] }));
  let ok = expected.length === 20;
  for (const e of expected) {
    if (!previews.find(p => p.file === e.file && p.line === e.line && p.at === head && p.event === event)) { ok = false; console.log("MISSING native preview", e.file, e.line); }
    const ext = external.find(a => a.href === e.url && a.label === e.label);
    if (!ext) { ok = false; console.log("MISSING external anchor", e.url); }
    if (e.pin !== "0afa98aa26be6756682f28defb081043ecc8efa2") { ok = false; console.log("WRONG pin", e.url); }
  }
  const unavailable = (html.match(/link unavailable/g) || []).length;
  const externalOK = external.every(a => /^https:\/\/github\.com\/generalbusiness-ai\/gitseq\/blob\/0afa98aa26be6756682f28defb081043ecc8efa2\//.test(a.href) && a.href.includes('rel="noopener noreferrer"') === false);
  console.log(JSON.stringify({ head, expectedCitations: expected.length, nativePreviewAnchors: previews.length, externalAnchors: external.length, unavailableMarkers: unavailable, allPaired: ok, externalAllPinnedGithub: externalOK, targetBlank: (html.match(/target="_blank" rel="noopener noreferrer"/g) || []).length }, null, 1));
  const filePreviews = previews.filter(p => !p.line); process.exitCode = ok && unavailable === 0 && previews.length === 21 && filePreviews.length === 1 && filePreviews[0].file === "internal/workroom/fold.go" && external.length === 20 ? 0 : 1;
} finally { await vite.close(); }
