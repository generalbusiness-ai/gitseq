import { readFileSync } from "node:fs";
import React from "react";
import { renderToStaticMarkup } from "react-dom/server";
import { createServer } from "vite";
const [uiRoot, notePath, head, event] = process.argv.slice(2);
const text = readFileSync(notePath, "utf8");
const vite = await createServer({ root: uiRoot, appType: "custom", logLevel: "silent", server: { middlewareMode: true } });
try {
  const { MarkdownPreview, PreviewContext } = await vite.ssrLoadModule("/src/components/Preview.tsx");
  const html = renderToStaticMarkup(React.createElement(PreviewContext.Provider, { value: { address: { kind: "notes", event }, open() {} } },
    React.createElement(MarkdownPreview, { event, commit: head, path: "notes/2026-09-07-previous-frontier-readers.md", text })));
  const anchors = [...html.matchAll(/<a href="([^"]+)"[^>]*>(.*?)<\/a>/g)].map(m => ({ href: m[1].replace(/&amp;/g, "&"), label: m[2] }));
  const previews = anchors.filter(a => a.href.includes("file=")).map(a => { const q = new URLSearchParams(a.href.slice(a.href.indexOf("?") + 1)); return { label: a.label, file: q.get("file"), at: q.get("at"), line: q.get("line"), event: q.get("preview_event") }; });
  const expected = [...text.matchAll(/`((?:internal|ui|cmd)\/[^`]+):(\d+)`/g)].map(m => ({ file: m[1], line: m[2] }));
  const unavailable = (html.match(/link unavailable/g) || []).length;
  let ok = true;
  for (const e of expected) {
    const hit = previews.find(p => p.file === e.file && p.line === e.line && p.at === head && p.event === event);
    if (!hit) { ok = false; console.log("MISSING preview link for", e); }
  }
  console.log(JSON.stringify({ head, expected: expected.length, previewAnchors: previews.length, allExpectedRendered: ok, unavailableRelativeLinks: unavailable, anchorsTotal: anchors.length }, null, 1));
  console.log("sample:", JSON.stringify(previews[0]));
  const external = anchors.filter(a => !a.href.startsWith("#/"));
  console.log("non-app anchors:", external.length);
  process.exitCode = ok && previews.length === expected.length ? 0 : 1;
} finally { await vite.close(); }
